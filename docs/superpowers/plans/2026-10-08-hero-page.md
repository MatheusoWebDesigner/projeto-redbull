# Hero RB22 — Página (scroll de vídeo) Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Construir a primeira seção da landing do RB22: um "vídeo" (sequência de quadros WebP) que avança e volta com a rolagem, com 4 capítulos de texto na coluna esquerda, carregamento em forma de largada de F1 (5 luzes) e telemetria ilustrativa.

**Architecture:** Vite + TypeScript sem framework. A lógica é dividida em módulos puros e testáveis (mapear rolagem → quadro/capítulo, ordem de carregamento, enquadramento "cover" no canvas, telemetria, luzes) e um módulo de integração (`hero.ts`) que liga GSAP ScrollTrigger, SplitText, Lenis e o DOM. O HTML já contém todo o texto (funciona e é lido por leitor de tela mesmo sem JS); o JS só anima.

**Tech Stack:** Vite, TypeScript, GSAP 3 (ScrollTrigger + SplitText, gratuitos desde a 3.13), Lenis (rolagem suave), Vitest (testes unitários).

**Spec:** `docs/superpowers/specs/2026-10-07-hero-design.md`

**Assets de entrada (gerados pelo blender-3d-specialist):**
- `assets/hero/frames/desktop/f_001.webp … f_240.webp` (1280 px de largura, 16:9)
- `assets/hero/frames/mobile/f_001.webp … f_240.webp` (720 px de largura, 9:16)
- `assets/hero/frames/poster_desktop.webp`, `assets/hero/frames/poster_mobile.webp` (quadro 240)

export const CHAPTER_COUNT = 4;
export const SNAP_POINTS = [0, 0.25, 0.5, 0.75, 1];

export function clamp01(p: number): number {
  return Math.min(1, Math.max(0, p));
}

export function frameIndexFor(progress: number, frameCount = FRAME_COUNT): number {
  return Math.round(clamp01(progress) * (frameCount - 1));

export function chapterIndexFor(progress: number, chapterCount = CHAPTER_COUNT): number {
  return Math.min(chapterCount - 1, Math.floor(clamp01(progress) * chapterCount));
}

import { FRAME_COUNT } from './scrub';

export type FrameSet = 'desktop' | 'mobile';

export function frameUrl(set: FrameSet, index: number): string {
  const n = String(index + 1).padStart(3, '0');
  return `/hero/frames/${set}/f_${n}.webp`;
}

export function splitLoadOrder(count: number, stride = 4): { priority: number[]; rest: number[] } {
  const priority: number[] = [];
  const rest: number[] = [];
  for (let i = 0; i < count; i++) {
    (i % stride === 0 || i === count - 1 ? priority : rest).push(i);
  }
  return { priority, rest };
}

export function nearestLoaded(loaded: ReadonlySet<number>, wanted: number, count: number): number | null {
  if (loaded.has(wanted)) return wanted;
  for (let d = 1; d < count; d++) {
    if (loaded.has(wanted - d)) return wanted - d;
    if (loaded.has(wanted + d)) return wanted + d;
  }
  return null;
}

const BATCH = 6;

export class FrameStore {
  private readonly images: (HTMLImageElement | null)[];
  private readonly loaded = new Set<number>();

  constructor(private readonly set: FrameSet, private readonly count = FRAME_COUNT) {
    this.images = new Array(count).fill(null);
  }

  /** Resolve quando os quadros prioritários carregam; o resto continua em segundo plano. */
  async load(onPriorityProgress: (fraction: number) => void): Promise<void> {
    const { priority, rest } = splitLoadOrder(this.count);
    let done = 0;
    for (let k = 0; k < priority.length; k += BATCH) {
      await Promise.all(
        priority.slice(k, k + BATCH).map(async (i) => {
          await this.loadOne(i);
          done++;
          onPriorityProgress(done / priority.length);
        }),
      );
    }
    void (async () => {
      for (let k = 0; k < rest.length; k += BATCH) {
        await Promise.all(rest.slice(k, k + BATCH).map((i) => this.loadOne(i)));
      }
    })();
  }

  get(index: number): HTMLImageElement | null {
    const i = nearestLoaded(this.loaded, index, this.count);
    return i === null ? null : this.images[i];
  }

  private loadOne(i: number): Promise<void> {
    return new Promise((resolve) => {
      const img = new Image();
      img.decoding = 'async';
      img.onload = () => {
        this.images[i] = img;
        this.loaded.add(i);
        resolve();
      };
      img.onerror = () => resolve();
      img.src = frameUrl(this.set, i);
    });
  }
}
```

export interface Rect {
  x: number;
  y: number;
  w: number;
  h: number;
}

/** Retângulo para desenhar a imagem cobrindo o canvas. focusX: 0 = mantém a esquerda, 1 = mantém a direita. */
export function coverRect(srcW: number, srcH: number, dstW: number, dstH: number, focusX = 0.5): Rect {
  const scale = Math.max(dstW / srcW, dstH / srcH);
  const w = srcW * scale;
  const h = srcH * scale;
  return { x: (dstW - w) * focusX, y: (dstH - h) / 2, w, h };
}

import { clamp01 } from './scrub';

export const TOP_SPEED = 320;
export const GEARS = 8;
/** Quadro 45 de 240 (índice 44 de 239): fim do carro parado. */
export const LAUNCH_PROGRESS = 44 / 239;

const IDLE_RPM = 4500;
const SHIFT_LOW = 10500;
const SHIFT_HIGH = 15000;

export function speedKmh(progress: number): number {
  const t = (clamp01(progress) - LAUNCH_PROGRESS) / (1 - LAUNCH_PROGRESS);
  if (t <= 0) return 0;
  const eased = t * t * (3 - 2 * t); // smoothstep
  return Math.round(TOP_SPEED * eased);
}

function gearPosition(progress: number): number {
  return (speedKmh(progress) / TOP_SPEED) * GEARS;
}

export function gear(progress: number): string {
  const g = gearPosition(progress);
  if (g === 0) return 'N';
  return String(Math.min(GEARS, Math.floor(g) + 1));
}

export function rpm(progress: number): number {
  const g = gearPosition(progress);
  if (g === 0) return IDLE_RPM;
  const within = g >= GEARS ? 1 : g - Math.floor(g);
  return Math.round(SHIFT_LOW + (SHIFT_HIGH - SHIFT_LOW) * within);
}
```

```ts
import { gsap } from 'gsap';
import { ScrollTrigger } from 'gsap/ScrollTrigger';
import { SplitText } from 'gsap/SplitText';
import Lenis from 'lenis';
import { coverRect } from './cover';
import { FrameStore, type FrameSet } from './frames';
import { litLights } from './lights';
import { chapterIndexFor, frameIndexFor, SNAP_POINTS } from './scrub';
import { gear, rpm, speedKmh } from './telemetry';

gsap.registerPlugin(ScrollTrigger, SplitText);

const FOCUS_X: Record<FrameSet, number> = { desktop: 0.7, mobile: 0.5 };

function qs<T extends Element>(sel: string): T {
  const el = document.querySelector<T>(sel);
  if (!el) throw new Error(`Elemento não encontrado: ${sel}`);
  return el;
}

function setupCanvas(canvas: HTMLCanvasElement): CanvasRenderingContext2D {
  const ctx = canvas.getContext('2d');
  if (!ctx) throw new Error('Canvas 2D indisponível');
  const resize = () => {
    const dpr = Math.min(window.devicePixelRatio || 1, 2);
    canvas.width = Math.round(canvas.clientWidth * dpr);
    canvas.height = Math.round(canvas.clientHeight * dpr);
  };
  resize();
  window.addEventListener('resize', resize);
  return ctx;
}

function draw(ctx: CanvasRenderingContext2D, img: HTMLImageElement, focusX: number): void {
  const { width, height } = ctx.canvas;
  const r = coverRect(img.naturalWidth, img.naturalHeight, width, height, focusX);
  ctx.clearRect(0, 0, width, height);
  ctx.drawImage(img, r.x, r.y, r.w, r.h);
}

function loadPoster(set: FrameSet): Promise<HTMLImageElement> {
  return new Promise((resolve, reject) => {
    const img = new Image();
    img.onload = () => resolve(img);
    img.onerror = reject;
    img.src = `/hero/frames/poster_${set}.webp`;
  });
}

function updateLoader(loader: HTMLElement, fraction: number): void {
  const lit = litLights(fraction);
  loader.querySelectorAll('.loader__lights span').forEach((s, i) => s.classList.toggle('is-lit', i < lit));
  loader.setAttribute('aria-valuenow', String(Math.round(fraction * 100)));
}

async function lightsOut(loader: HTMLElement): Promise<void> {
  updateLoader(loader, 1);
  await new Promise((r) => setTimeout(r, 450));
  loader.querySelectorAll('.loader__lights span').forEach((s) => s.classList.remove('is-lit'));
  loader.classList.add('is-done');
  loader.setAttribute('aria-hidden', 'true');
}

export async function initHero(): Promise<void> {
  const set: FrameSet = window.matchMedia('(max-aspect-ratio: 1/1)').matches ? 'mobile' : 'desktop';
  const reduced = window.matchMedia('(prefers-reduced-motion: reduce)').matches;
  const canvas = qs<HTMLCanvasElement>('.hero__canvas');
  const loader = qs<HTMLElement>('.loader');
  const ctx = setupCanvas(canvas);

  if (reduced) {
    document.documentElement.classList.add('reduced');
    const poster = await loadPoster(set);
    draw(ctx, poster, FOCUS_X[set]);
    window.addEventListener('resize', () => draw(ctx, poster, FOCUS_X[set]));
    loader.classList.add('is-done');
    loader.setAttribute('aria-hidden', 'true');
    return;
  }

  const store = new FrameStore(set);
  await store.load((f) => updateLoader(loader, f));
  await lightsOut(loader);

  const lenis = new Lenis();
  lenis.on('scroll', ScrollTrigger.update);
  gsap.ticker.add((time) => lenis.raf(time * 1000));
  gsap.ticker.lagSmoothing(0);

  const chapters = Array.from(document.querySelectorAll<HTMLElement>('.chapter'));
  const bars = Array.from(document.querySelectorAll<HTMLElement>('.hero__progress span'));
  const speedEl = qs<HTMLElement>('[data-speed]');
  const rpmEl = qs<HTMLElement>('[data-rpm]');
  const gearEl = qs<HTMLElement>('[data-gear]');
  const splits = chapters.map((c) => SplitText.create(c.querySelector('.chapter__title'), { type: 'chars' }));

  let currentFrame = -1;
  let currentChapter = -1;

  const showChapter = (index: number) => {
    if (index === currentChapter) return;
    const prev = chapters[currentChapter];
    if (prev) gsap.to(prev, { autoAlpha: 0, y: -24, duration: 0.3, ease: 'power2.in', onComplete: () => prev.classList.remove('is-active') });
    const next = chapters[index];
    next.classList.add('is-active');
    gsap.fromTo(next, { autoAlpha: 0, y: 24 }, { autoAlpha: 1, y: 0, duration: 0.6, ease: 'power3.out' });
    gsap.from(splits[index].chars, { yPercent: 110, opacity: 0, duration: 0.6, stagger: 0.025, ease: 'power3.out' });
    bars.forEach((b, i) => b.classList.toggle('is-on', i <= index));
    currentChapter = index;
  };

  const render = (progress: number) => {
    const frame = frameIndexFor(progress);
    if (frame !== currentFrame) {
      const img = store.get(frame);
      if (img) draw(ctx, img, FOCUS_X[set]);
      currentFrame = frame;
    }
    showChapter(chapterIndexFor(progress));
    speedEl.textContent = String(speedKmh(progress));
    rpmEl.textContent = String(rpm(progress));
    gearEl.textContent = gear(progress);
  };

  ScrollTrigger.create({
    trigger: '.hero',
    start: 'top top',
    end: '+=400%',
    pin: '.hero__stage',
    scrub: true,
    snap: { snapTo: SNAP_POINTS, duration: { min: 0.2, max: 0.6 }, ease: 'power1.inOut' },
    onUpdate: (self) => render(self.progress),
    onRefresh: (self) => {
      currentFrame = -1;
      render(self.progress);
    },
  });
  render(0);
}
```
